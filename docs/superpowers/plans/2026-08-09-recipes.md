```swift
public struct Recipe: Codable, Equatable, Sendable, Identifiable {
    public var id: UUID
    public var name: String
    public var iso: Float?
    public var shutterSeconds: Double?
    public var evBias: Float
    public var wbKelvin: Int?
    public var raw: RAWMode
    public var heifCompanion: Bool
    public var processing: ProcessingLevel
    public var aspectIndex: Int
    public init(id: UUID = UUID(), name: String, iso: Float?, shutterSeconds: Double?,
                evBias: Float, wbKelvin: Int?, raw: RAWMode, heifCompanion: Bool,
                processing: ProcessingLevel, aspectIndex: Int)
}

public struct RecipeBook: Equatable, Sendable {
    public private(set) var recipes: [Recipe]
    public init() // empty
    public init(data: Data) // JSON decode; corrupt/invalid -> empty
    public func encode() -> Data // JSON
    public mutating func add(_ recipe: Recipe) // same-name (case-sensitive) replaces in place, else append
    public mutating func remove(id: UUID)
}
```

`RAWMode` and `ProcessingLevel` are String-raw enums — make them Codable via conformance declaration if not already (check; String-raw enums get Codable for free once declared `Codable`; add retroactive conformance in Recipes.swift ONLY if they don't already conform).

**Authoritative behavior:**

- Apply function:

```swift
    private func apply(_ recipe: Recipe) {
        engine.setISO(recipe.iso)
        engine.setShutter(recipe.shutterSeconds)
        engine.setEVBias(recipe.evBias)
        engine.setWhiteBalance(kelvin: recipe.wbKelvin)
        engine.formatSelection = FormatSelection(raw: recipe.raw, heifCompanion: recipe.heifCompanion)
        engine.processingLevel = recipe.processing
        if AspectRatio.allCases.indices.contains(recipe.aspectIndex) {
            aspect = AspectRatio.allCases[recipe.aspectIndex]
        }
        Haptics.success()
        chromeInteraction()
    }
```
